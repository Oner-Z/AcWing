class Solution {
public:
    bool isPalindrome(string s) {
        string s_l;
        for (auto c : s) {
            if (isalnum(c)) {
                s_l += tolower(c);
            }
        }
        int n = s_l.size();
        int l = 0, r = n - 1;
        while (l < r) {
            if (s_l[l] != s_l[r]) return false;
            l ++ ;
            r -- ;
        }
        return true;
    }
};
