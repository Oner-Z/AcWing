class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        int s = numbers.size();
        vector<int> res;
        int l = 0, r = s - 1;
        while (l < r) {
            if (numbers[l] + numbers[r] > target) {
                r -- ;
            }
            else if (numbers[l] + numbers[r] < target) {
                l ++ ;
            }
            else {
                res.push_back(l + 1);
                res.push_back(r + 1);
                return res;
            }
        }
        return {-1, -1};
    }
};
