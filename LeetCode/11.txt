class Solution {
public:
    int maxArea(vector<int>& height) {
        int s = height.size();
        int l = 0, r = s - 1;
        int ans = -1;
        while (l < r) {
            int tmp = min(height[l], height[r]) * (r - l);
            ans = max(ans, tmp);
            if (height[l] < height[r]) {
                l ++ ;
            }
            else {
                r -- ;
            }
        }
        return ans;
    }
};
