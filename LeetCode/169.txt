class Solution {
public:
    int majorityElement(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        return nums[nums.size() / 2];
    }
};

class Solution {
public:
    int majorityElement(vector<int>& nums) {
        unordered_map<int, int> mp;
        int max = -1, res = 0;
        for (auto num: nums) {
            mp[num] ++ ;
            if (mp[num] > max){
                max = mp[num];
                res = num;
            }
        }
        return res;
    }
};
