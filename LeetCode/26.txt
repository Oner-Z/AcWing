class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int s = nums.size();
        int l = 0, r = 1;
        while (r < s){
            if (nums[l] != nums[r]){
                l ++ ;
                nums[l] = nums[r];
            }
            r ++ ;
        }
        return l + 1;
    }
};
